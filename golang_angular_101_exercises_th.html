โจทย์ที่ 3: ระบบสแกนหาเลขคี่ในกล่องข้อมูล (Loops & Modulo) Go 101

โจทย์: จงเขียนฟังก์ชัน CountOddNumbers(limit int) int คอยวนลูปตั้งแต่เลข 1 ถึงตัวแปรเกณฑ์ limit แล้วนับว่ามี "จำนวนคี่" ทั้งหมดกี่ตัว และคืนค่ากลับไป

เช็คเลขคี่โดยการนำตัวเลขในลูป i % 2 != 0 และขยับตัวนับสะสมบวกเพิ่มสะสม

🔎 คลิกเพื่อดูเฉลยข้อที่ 3 
func CountOddNumbers(limit int) int {
	count := 0
	for i := 1; i <= limit; i++ {
		if i % 2 != 0 {
			count++
		}
	}
	return count
}

โจทย์ที่ 5: สมุดจดดึงเบอร์โทรศัพท์ร้านค้าด่วน (Maps & Lookups) Go 101

โจทย์: จงเขียนฟังก์ชัน FindPhone(directory map[string]string, name string) string เพื่อค้นหาเบอร์โทรศัพท์ของคนชื่อ name ในแผนที่ หากค้นหาไม่พบให้ส่งข้อความ "Not Found" คืนกลับไป

ใช้ตัวแปรตรวจจับการมีอยู่ val, ok := directory[name] หาก ok เป็น false ให้รีเทิร์นส่งคำเตือนออกไป

🔎 คลิกเพื่อดูเฉลยข้อที่ 5 
func FindPhone(directory map[string]string, name string) string {
	phone, exists := directory[name]
	if !exists {
		return "Not Found"
	}
	return phone
}

func DivideNumbers(a, b float64) (float64, error) {
	if b == 0 {
		return 0, errors.New("cannot divide by zero")
	}
	return a / b, nil
}

โจทย์ที่ 9: ตู้นิรภัยฝากสะสมของปลอดภัยขนานกัน (Mutex Safe Counter) Go 101

โจทย์: จงนิยาม Struct ชื่อ SafeCounter เก็บตัวแปร value int และมี mu sync.Mutex อยู่ในตัว พร้อมเขียนเมทอด (s *SafeCounter) Add() ควบคุมคิวปลอดภัย

เรียกใช้ s.mu.Lock() เพื่อกันคนอื่นเข้ามาแก้ไขค่าแรม และสั่งคลายล็อคเมื่อจบฟังก์ชัน

🔎 คลิกเพื่อดูเฉลยข้อที่ 9 
import "sync"

type SafeCounter struct {
	mu    sync.Mutex
	value int
}

func (s *SafeCounter) Add() {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.value++
}

โจทย์ที่ 18: การบันทึกและพ่นแจ้งเตือนเมื่อค่าขยับ (Signals Effect) Angular 101

export class LoggerComp {

โจทย์ที่ 21: ฟอร์มสร้างของใช้ใหม่ส่งบันทึกหลังบ้าน (POST Add Item) Full-Stack

โจทย์: จงแสดงสเต็ปโค้ดการทำแบบฟอร์มเพิ่มสินค้าใหม่จากหน้าบ้าน Angular ส่งผ่านคำขอแบบ POST พร้อม JSON Body ไปยัง Go เพื่อให้ Go รับข้อมูล แกะรหัสสร้างวัตถุชิ้นใหม่บันทึกเก็บใส่แรมแบบปลอดภัย

ฝั่ง Go ใช้คำสั่งตรวจสอบไวด์ข้อมูล c.ShouldBindJSON(&input) เพื่อแกะถอดรหัส และแอดเข้า Repository

🔎 คลิกเพื่อดูเฉลยข้อที่ 21

ฝั่ง Go Backend (`item_handler.go`):

func (h *ItemHandler) Create(c *gin.Context) {
	var input CreateInput
	// แกะถอด JSON ที่ส่งมาทางเน็ต
	if err := c.ShouldBindJSON(&input); err != nil {
		c.JSON(400, gin.H{"error": err.Error()})
		return
	}
	
	newItem := models.InventoryItem{
		ID: "item_" + fmt.Sprintf("%d", time.Now().UnixNano()),
		Name: input.Name,
		CurrentStock: input.CurrentStock,
	}
	saved, _ := h.useCase.Create(newItem)
	c.JSON(201, saved)
}

ฝั่ง Angular Frontend (`api.ts`):

createItem(item: any): Observable<InventoryItem> {

  return this.http.post<InventoryItem>('http://localhost:8080/api/items', item);
}

โจทย์ที่ 22: สัญญารับฟันลบข้อมูลออกจากระบบ (DELETE Deletion) Full-Stack

โจทย์: จงสรุปขั้นตอนการทำปุ่มลบของใช้ออกจากจอ ที่เมื่อคลิกแล้วจะส่งรหัส ID ผ่าน HTTP แบบ DELETE ไปหลังบ้าน Go เพื่อลบออกจากตู้แผนที่ และเคลียร์ตัวแปรในการ์ดหน้าบ้านให้สลายตัวหายไปทันทีโดยไม่ต้อง F5 รีโหลดแผ่นหน้าเว็บ

หลังลบจาก API สำเร็จ ให้แก้ไขสเตตัส Signal กลางด้วยการคัดกรองฟิลเตอร์ this._items.set(this._items().filter(item => item.id !== id))

🔎 คลิกเพื่อดูเฉลยข้อที่ 22

ฝั่ง Go Backend (`item_handler.go`):

api.DELETE("/items/:id", func(c *gin.Context) {
	id := c.Param("id")
	itemRepo.Delete(id) // ลบออกจาก RAM
	c.JSON(200, gin.H{"message": "Deleted"})
})

ฝั่ง Angular Frontend (`inventory.service.ts`):

removeItem(id: string) {
  this.api.deleteItem(id).subscribe(() => {
    // ⭐️ กรองสลัดตัดไอเทมที่ถูกสั่งลบออกจากสัญญาณ Signal ทันที!
    const remaining = this._items().filter(item => item.id !== id);
    this._items.set(remaining); // หน้าจอจะสลายการ์ดทิ้งทันทีภายในพริบตา!
  });
}
